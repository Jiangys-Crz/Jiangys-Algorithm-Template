% ---------- 字体 ----------
% 按平台选择字体。这里用“字体文件是否存在”来判断平台，而不是 \IfFontExistsTF：
% 后者在字体确实缺失时会让 XeTeX 把字体名当文件名去找，触发 mktextfm 报错。
% macOS
\IfFileExists{/System/Library/Fonts/Supplemental/Songti.ttc}{%
    \setmainfont{Times New Roman}
    \setCJKmainfont{Songti SC}[AutoFakeBold=2]
    \setCJKfamilyfont{zhsong}{Songti SC}[AutoFakeBold=2]
    \setCJKsansfont{PingFang SC}[AutoFakeBold=2]
    \setCJKfamilyfont{zhhei}{PingFang SC}[AutoFakeBold=2]
    \setmonofont{Menlo}
}{%
    % Windows
    \IfFileExists{C:/Windows/Fonts/simsun.ttc}{%
        \setmainfont{Times New Roman}
        \setCJKmainfont{SimSun}[AutoFakeBold=2]
        \setCJKfamilyfont{zhsong}{SimSun}[AutoFakeBold=2]
        \setCJKsansfont{SimHei}[AutoFakeBold=2]
        \setCJKfamilyfont{zhhei}{SimHei}[AutoFakeBold=2]
        \setmonofont{Consolas}
    }{%
        % Linux / 其它
        \setmainfont{Noto Serif}
        \setCJKmainfont{Noto Serif CJK SC}[AutoFakeBold=2]
        \setCJKfamilyfont{zhsong}{Noto Serif CJK SC}[AutoFakeBold=2]
        \setCJKsansfont{Noto Sans CJK SC}[AutoFakeBold=2]
        \setCJKfamilyfont{zhhei}{Noto Sans CJK SC}[AutoFakeBold=2]
        \setmonofont{DejaVu Sans Mono}
    }%
}

% book 类默认 secnumdepth=2、tocdepth=2，四级标题（subsubsection）既不编号也不进目录。
% 这里都放到 3，让它带编号（形如 1.1.1.1）并出现在目录中。
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{3}

% 保持小节编号仍然是 4.1、4.1.1，而不是 四.1
